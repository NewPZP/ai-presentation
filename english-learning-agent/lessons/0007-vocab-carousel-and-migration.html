// 推荐的下一个功能（不强制，用户可自由跳转）
export function getNextFeature(current: Feature): Feature | null {
  const index = FEATURE_ORDER.indexOf(current);
  if (index === -1 || index === FEATURE_ORDER.length - 1) return null;
  return FEATURE_ORDER[index + 1];
}

export function getFeatureIndex(feature: Feature): number {
  return FEATURE_ORDER.indexOf(feature) + 1;
}

export const TOTAL_FEATURES = FEATURE_ORDER.length;

// 三态状态（和旧 PhaseStatus 完全一样，只改名）
export type FeatureStatus = "notStarted" | "inProgress" | "completed";

// Article 接口（featureStatus 替代 phaseStatus）
export interface Article {
  id: string;
  fileName: string;
  text: string;
  wordCount: number;
  importedAt: number;
  featureStatus: Record<Feature, FeatureStatus>;
}

export function createDefaultFeatureStatus(): Record<Feature, FeatureStatus> {
  return {
    vocab: "notStarted",
    podcast: "notStarted",
    listening: "notStarted",
    vocabQuiz: "notStarted",
    finalQuiz: "notStarted",
    progress: "notStarted",
  };
}

export function getProgress(article: Article): number {
  const completed = FEATURE_ORDER.filter(
    (f) => article.featureStatus[f] === "completed"
  ).length;
  return Math.round((completed / TOTAL_FEATURES) * 100);
}

CSS 3D 翻转原理

翻转效果靠 4 个 CSS 属性配合：

属性作用
perspective: 1000px父容器设透视距离，让 3D 旋转有近大远小效果
transform-style: preserve-3d内核元素保留 3D 空间，子元素才能在 3D 中定位
backface-visibility: hidden背面隐藏——正面转过去后看不见，背面的背面也看不见
transform: rotateY(180deg)Y 轴旋转 180°，正面转到背面、背面转到正面

动手指引：在浏览器里试试——把上面的代码跑起来后，用 DevTools 检查元素，手动在内核 div 上加/去 rotateY(180deg)，你能看到卡片翻转的过渡动画。transition-transform duration-500 控制翻转速度（500ms）。

翻转卡片的交互逻辑

正面（单词 + 音标 + 🔊）
  ├── 点击卡片 → 翻转
  ├── 点 🔊 → 播放发音（不翻转，stopPropagation）
  ├── 点「不认识」→ onUnknown() → 翻转看背面
  └── 点「认识」→ onKnown() → 移出队列

背面（释义 + 例句 + 翻译）
  ├── 点击卡片 → 翻回正面
  └── 点「翻回正面」→ 翻回

第六步：间隔重复队列

翻转卡片做好了，现在写轮播逻辑——间隔重复。

新建 app/articles/[id]/[feature]/vocab-view.tsx（词汇卡片轮播子组件）：

// app/articles/[id]/[feature]/vocab-view.tsx
"use client";

import { useState, useEffect } from "react";
import { VocabFlipCard } from "@/components/vocab-flip-card";
import { VocabWord } from "@/lib/schemas";
import { Loader2, BookOpen, CheckCircle2 } from "lucide-react";
import { Button } from "@/components/ui/button";

export function VocabView({
  articleText,
  onComplete,
}: {
  articleText: string;
  onComplete: () => void;  // 全部学完的回调
}) {
  const [words, setWords] = useState<VocabWord[]>([]);
  const [queue, setQueue] = useState<VocabWord[]>([]);  // 待学队列
  const [knownCount, setKnownCount] = useState(0);
  const [loading, setLoading] = useState(true);

  // 提取生词（复用第 6 课的 API）
  useEffect(() => {
    fetch("/api/extract-vocab", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ articleText }),
    })
      .then((res) => res.json())
      .then((data) => {
        setWords(data.words);
        setQueue(data.words);  // 初始队列 = 全部单词
        setLoading(false);
      })
      .catch((err) => {
        console.error(err);
        setLoading(false);
      });
  }, [articleText]);

  // 认识 → 移出队列
  function handleKnown() {
    setQueue((prev) => prev.slice(1));  // 移除队首
    setKnownCount((prev) => prev + 1);
  }

  // 不认识 → 翻转看背面，然后移到队尾
  function handleUnknown() {
    setQueue((prev) => {
      const [first, ...rest] = prev;
      return [...rest, first];  // 队首移到队尾
    });
  }

  // 加载中
  if (loading) {
    return (
      <div className="flex items-center justify-center py-12">
        <Loader2 className="w-6 h-6 animate-spin text-muted-foreground" />
      </div>
    );
  }

  // 全部学完
  if (queue.length === 0) {
    return (
      <div className="text-center py-12 space-y-3">
        <CheckCircle2 className="w-12 h-12 mx-auto text-primary" />
        <h3 className="text-lg font-medium">词汇预习完成！</h3>
        <p className="text-muted-foreground">
          共 {words.length} 个词，你认识了 {knownCount} 个
        </p>
        <Button onClick={onComplete}>继续下一个功能</Button>
      </div>
    );
  }

  // 当前卡片
  const currentWord = queue[0];

  return (
    <div className="space-y-4">
      {/* 进度 */}
      <div className="text-center text-sm text-muted-foreground">
        进度：{knownCount}/{words.length} 已认识
      </div>

      {/* 进度条 */}
      <div className="flex gap-1 justify-center">
        {words.map((_, i) => (
          <div
            key={i}
            className={cn(
              "w-2 h-2 rounded-full",
              i < knownCount ? "bg-primary" : "bg-border"
            )}
          />
        ))}
      </div>

      {/* 翻转卡片 */}
      <VocabFlipCard
        key={currentWord.word}  // ★ key 变化触发组件重置（翻转状态归零）
        word={currentWord}
        onKnown={handleKnown}
        onUnknown={handleUnknown}
      />
    </div>
  );
}

import { cn } from "@/lib/utils";

间隔重复的核心逻辑

初始队列：[A, B, C, D, E]

认识 A → 移出队列：[B, C, D, E]        ← A 不再出现
不认识 B → 移到队尾：[C, D, E, B]       ← B 会再出现

最终队列空了 → 全部认识 → 完成

这是最简间隔重复——不认识就重新排队，认识就移出。真正的 Anki 用的是 SM-2 算法（按遗忘曲线安排复习时间），但队列式已经够用，且容易理解。

关键细节：key={currentWord.word}。React 通过 key 判断组件是否复用。如果不加 key，从卡片 A 切到卡片 B 时，VocabFlipCard 组件实例不变，flipped 状态会保留——你可能在翻转状态下看到下一张卡。加 key 后，卡片切换 = 旧组件卸载 + 新组件挂载，flipped 自动重置为 false。

"use client";

import { use } from "react";
import { useRouter } from "next/navigation";
import Link from "next/link";
import { useState, useEffect } from "react";
import { useArticles } from "@/hooks/use-articles";
import {
  Feature,
  FEATURE_INFO,
  FEATURE_ORDER,
  getNextFeature,
  getFeatureIndex,
  TOTAL_FEATURES,
} from "@/lib/learning-features";
import { Button } from "@/components/ui/button";
import { ScrollArea } from "@/components/ui/scroll-area";
import { ArrowLeft, ChevronRight, Loader2 } from "lucide-react";
import { VocabView } from "./vocab-view";

function isValidFeature(f: string): f is Feature {
  return FEATURE_ORDER.includes(f as Feature);
}

export default function LearnPage({
  params,
}: {
  params: Promise<{ id: string; feature: string }>;
}) {
  const { id, feature } = use(params);
  const router = useRouter();
  const { loaded, getArticle, updateFeatureStatus } = useArticles();
  const [input, setInput] = useState("");

  const isValid = isValidFeature(feature);
  const currentFeature = (isValid ? feature : "vocab") as Feature;
  const article = getArticle(id);
  const nextFeature = getNextFeature(currentFeature);

  // 所有 Hooks 在 return 之前
  useEffect(() => {
    if (!isValid) router.push("/");
  }, [isValid, router]);

  useEffect(() => {
    if (loaded && !article) router.push("/");
  }, [loaded, article, router]);

  useEffect(() => {
    if (article && article.featureStatus[currentFeature] === "notStarted") {
      updateFeatureStatus(id, currentFeature, "inProgress");
    }
  }, [id, currentFeature, article, updateFeatureStatus]);

  if (!loaded) {
    return (
      <div className="flex items-center justify-center h-screen">
        <Loader2 className="w-6 h-6 animate-spin text-muted-foreground" />
      </div>
    );
  }

  if (!isValid || !article) return null;

  function goNext() {
    updateFeatureStatus(id, currentFeature, "completed");
    if (nextFeature) {
      router.push(`/articles/${id}/${nextFeature}`);
    }
  }

  function handleVocabComplete() {
    updateFeatureStatus(id, "vocab", "completed");
    if (nextFeature) {
      router.push(`/articles/${id}/${nextFeature}`);
    }
  }

  const info = FEATURE_INFO[currentFeature];

  return (
    <div className="flex flex-col h-screen max-w-2xl mx-auto px-4 py-4">
      {/* 顶部导航 */}
      <div className="flex items-center gap-3 mb-4 pb-3 border-b">
        <Link href="/">
          <Button variant="ghost" size="icon">
            <ArrowLeft className="w-4 h-4" />
          </Button>
        </Link>
        <div className="flex-1">
          <h2 className="font-medium">
            {info.label}
            <span className="text-muted-foreground font-normal ml-2">
              {getFeatureIndex(currentFeature)}/{TOTAL_FEATURES}
            </span>
          </h2>
          <p className="text-xs text-muted-foreground">{article.fileName}</p>
        </div>
        {nextFeature && (
          <Button variant="secondary" size="sm" onClick={goNext}>
            {FEATURE_INFO[nextFeature].label}
            <ChevronRight className="w-4 h-4 ml-1" />
          </Button>
        )}
      </div>

      {/* 功能内容区 */}
      <ScrollArea className="flex-1 pr-4">
        {currentFeature === "vocab" && (
          <VocabView
            articleText={article.text}
            onComplete={handleVocabComplete}
          />
        )}

        {currentFeature !== "vocab" && (
          <div className="text-center text-muted-foreground py-12">
            <p>{info.description}</p>
            <p className="text-sm mt-2">（此功能将在后续课程实现）</p>
          </div>
        )}
      </ScrollArea>
    </div>
  );
}

关键点：

  1. 所有 Hooks 在 return 前 — 第 6 课踩的坑记得避开。三个 useEffect + if (!loaded) + if (!isValid || !article) return null 顺序不能错。
  2. currentFeature === "vocab" 条件渲染 — 只有词汇卡片功能渲染 VocabView，其他功能显示「后续课程实现」占位。后续课程逐个填。
  3. handleVocabComplete 回调 — 词汇卡片全部学完时调用，标记 vocab 为 completed 并跳转下一个功能。
  4. 去掉了 useChat — 词汇卡片功能不需要聊天，聊天功能后续在需要的模块里再加。

小测验

3D 翻转原理

CSS 翻转卡片中 backface-visibility: hidden 的作用是什么？

  • 让卡片背面透明
  • 防止卡片翻转过快
  • 让正面转过去后隐藏，背面的背面也隐藏，只有当前面可见
  • 禁用卡片的点击事件

间隔重复队列

点「不认识」时，队列 [A,B,C] 会变成什么？

  • [B, C]（A 被移除）
  • [B, C, A]（A 移到队尾，会再出现）
  • [A, B, C]（不变）
  • [C, A, B]（随机打乱）

key 的作用

VocabFlipCard 加 key={currentWord.word} 的目的是什么？

  • 切换卡片时重置翻转状态（flipped 归零）
  • 让卡片渲染更快
  • 防止重复单词
  • 让 CSS 动画更流畅

数据驱动的回报

为什么这次迁移能「换数据不换逻辑」？

  • 因为 TypeScript 类型系统很强
  • 因为第 3 课把配置数据和转移逻辑分离了（数据驱动设计）
  • 因为新功能和旧功能完全一样
  • 因为 React 会自动处理迁移

主源推荐

👉 MDN · transform-style: preserve-3d

CSS 3D 变换的权威文档。perspective、transform-style、backface-visibility 三个属性配合实现翻转效果。本课的翻转卡片就是这个原理。

间隔重复算法：SuperMemo SM-2 算法（Wikipedia）。本课用的是最简队列式，真正的 Anki 用 SM-2 按遗忘曲线安排复习时间。第 14 课升级时参考。
